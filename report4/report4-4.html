let finished = false;
let alerted = false;

const users = ["User1", "User2", "User3"];
const passwords = ["password123", "1234", "Qwerty123"];

while(finished == false){
    alerted = false;
    let user = prompt("Enter your login");
    for (let i = 0; i < users.length; i++){
        if (user == users[i]){
            let password = prompt("Enter your password");
            if (password == passwords[i]){
                alert("Hello " + users[i]);
                finished = true;
                alerted = true;
            }
            else if (password == "" || password == null){
                alerted = true;
            }
            else if (password != passwords[i]){
                alert("I don't know you");
                alerted = true;
            }
        }
        else if (i == users.length-1 && alerted == false){
            alert("I don't know you");
        }
    }
}
