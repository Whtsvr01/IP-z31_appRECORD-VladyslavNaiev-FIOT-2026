function makeArray(firstArray, secondArray, maxLength){
    newArray = firstArray.concat(secondArray);
    if (newArray.length > maxLength){
        newArray = newArray.slice(0, maxLength);
    }
    return newArray;
}

let firstArray = prompt("Enter firstArray in a format 1,2,3,4");
firstArray = firstArray.split(",");
let secondArray = prompt("Enter secondArray in a format 1,2,3,4");
secondArray = secondArray.split(",");
let maxLength = prompt("Enter maxLength for the array");
let result = makeArray(firstArray, secondArray, maxLength);
alert("Result: " + result);
