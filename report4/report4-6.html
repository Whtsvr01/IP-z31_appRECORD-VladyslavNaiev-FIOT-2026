function makeTransaction(quantity, pricePerDroid, customerCredits){
    let totalPrice = parseInt(quantity) * parseFloat(pricePerDroid);
    if (totalPrice <= customerCredits){
        alert ("You ordered " + quantity + " droids worth " + totalPrice + " credits!")
    }
    else{
        alert("Insufficient funds!");
    }
}

let quantity = prompt("Enter quantity");
let pricePerDroid = prompt("Enter price");
let customerCredits = prompt("Enter credits");
makeTransaction(quantity, pricePerDroid, customerCredits);
