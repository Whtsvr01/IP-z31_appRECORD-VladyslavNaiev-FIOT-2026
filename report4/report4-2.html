let hour = prompt("enter time in a format – hours:minutes");
let pattern = /^\d\d:\d\d$/;
if (pattern.test(hour)){
    let results = hour.split(":");
    let hours = parseInt(results[0]);
    let minutes = parseInt(results[1]);

    if (hours > 24){
        alert("Hour value cannot be greater than 24");
    }
    else if (minutes > 60){
        alert("Minutes value cannot be greater than 60");
    }
    else{
        if (minutes => 15){
            alert("First quater");
        }
        else if (minutes => 30){
            alert("Second quater");
        }
        else if (minutes => 45){
            alert("Third quater");
        }
        else{
            alert("Fourth quater");
        }
    }
}
else{
    alert("Wrong value for the time");
}
