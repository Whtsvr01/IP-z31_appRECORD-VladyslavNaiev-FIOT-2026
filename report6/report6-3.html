Javascript


let box = document.getElementById("box");

let enlarge = document.getElementById("enlarge");
let shrink = document.getElementById("shrink");

enlarge.addEventListener("click", function(){
    let widthBox = parseInt(box.style.width);
    let heightBox = parseInt(box.style.height);

    if (widthBox <= 100){
        widthBox += 15;
        heightBox += 15;
    }

    box.style.width = widthBox + "px";
    box.style.height = heightBox + "px";
});

shrink.addEventListener("click", function(){
    let widthBox = parseInt(box.style.width);
    let heightBox = parseInt(box.style.height);

    if (widthBox > 0){
        widthBox -= 15;
        heightBox -= 15;
    }

    box.style.width = widthBox + "px";
    box.style.height = heightBox + "px";
});
