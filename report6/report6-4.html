double.addEventListener("click", function(){
    let numbers = document.querySelectorAll("#list li");
    const originalNumbers = [1, 4, 8, 16, 20, 30];
    if (numbers[numbers.length-1].textContent > 1000000000){
        numbers.forEach((number, i) => {
            number.textContent = originalNumbers[i];
        });
    }
    else{
        numbers.forEach(number => {
            number.textContent = parseInt(number.textContent) * 2;
        });
    }
});
