function checkBrackets(str){
    let currentBrackets = [];
    const brackets = {")": "(", "]": "[", "}": "{"};
    for (let i = 0; i < str.length; i++)
    {
        let element = str[i];
        if (Object.keys(brackets).includes(element)){
            if (currentBrackets[currentBrackets.length-1] == brackets[element]){
                currentBrackets.pop();
            }
            else{
                return false;
            }
        }
        else if (Object.values(brackets).includes(element)){
            currentBrackets.push(element);
        }
    };
    if (currentBrackets.length == 0){
        return true;
    }
    return false;
}
